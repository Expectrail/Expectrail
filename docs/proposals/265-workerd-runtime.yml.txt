# Proposal only. Sean applies it to CI; this is not a workflow or deployment edit.
# Local workerd + fake GitHub: no credentials, dispatch, deployed API or model.
name: Run it native runtime
on:
  pull_request:
  push:
    branches: [main]
permissions:
  contents: read
jobs:
  workerd-runtime:
    runs-on: ubuntu-24.04
    timeout-minutes: 8
    steps:
      - uses: actions/checkout@11bd71901bbe5b1630ceea73d27597364c9af683 # v4.2.2
        with:
          persist-credentials: false
      - uses: actions/setup-node@49933ea5288caeca8642d1e84afbd3f7d6820020 # v4.4.0
        with:
          node-version: '22'
      - uses: actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065 # v5.6.0
        with:
          python-version: '3.12'
      - name: Install contract reader and pinned runtime
        shell: bash
        run: |
          set -euo pipefail
          python3 -m pip install -e '.[contracts]'
          mkdir -p "$RUNNER_TEMP/run-it-runtime"
          cp apps/run-it-edge/runtime/package.json apps/run-it-edge/runtime/package-lock.json "$RUNNER_TEMP/run-it-runtime/"
          npm ci --prefix "$RUNNER_TEMP/run-it-runtime" --cache "$RUNNER_TEMP/npm-cache" --ignore-scripts --no-audit --no-fund
      - name: Require native Worker and polling checks
        shell: bash
        env:
          NODE_PATH: ${{ runner.temp }}/run-it-runtime/node_modules
          EXPECTRAIL_WORKER_TESTS: required
        run: |
          set -euo pipefail
          {
            node apps/run-it-edge/test/runtime-check.mjs
            python3 -m unittest -v tests.test_runit_status_bound
          } 2>&1 | tee workerd-runtime.log
      - name: Retain runtime log
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02 # v4.6.2
        with:
          name: workerd-runtime
          path: workerd-runtime.log
          if-no-files-found: warn
          retention-days: 14
